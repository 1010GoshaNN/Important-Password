<!-- JS-->
<script>
const box = document.getElementById('Box');
const status = document.getElementById('newStatus');

function NewPassword() {
    const savedPass = localStorage.getItem('SinglePassword');

    if (savedPass) {
        box.innerText = savedPass;
    } else {
        box.innerText = "Nothing to show";
    }
}

function Password() {
    let string = prompt("Enter new password");

    if (string !== null && string.trim() !== "") {
        if (string.indexOf(' ') == 0 || string.includes('%')) {
            alert("Error: Space and % are not allowed");
            return;
        }
        localStorage.setItem('SinglePassword', string.trim());
        box.innerText = string.trim();
        status.innerText = "Password is up to date";
        status.style.color = "#38a168";

    setTimeout(() => {
        status.innerText = " New Password";
        status.style.color = "#a0aec0";
    }, 2000);
    } else if (string !== null) {
        alert("Sorry, bur Password should be filled!")
    }
}

NewPassword();
</script>
